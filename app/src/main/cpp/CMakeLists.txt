# Builds PRoot (GPL-2.0-or-later, Termux fork, git submodule third_party/proot) and its talloc
# dependency (LGPL-3.0-or-later, vendored in third_party/talloc) from source.
#
# Android only runs libraries named lib*.so from nativeLibraryDir, so the two executables are
# produced with that name: libproot.so (the proot binary) and libproot-loader.so (the static
# loader proot injects into traced programs, found at runtime through PROOT_LOADER).

cmake_minimum_required(VERSION 3.22)
project(ultimateterminal_native C ASM)

set(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../..")
set(PROOT_SRC "${REPO_ROOT}/third_party/proot/src")
set(TALLOC_SRC "${REPO_ROOT}/third_party/talloc")
set(PROOT_VERSION "5.1.107.96")

# Per-ABI values copied from proot's src/arch.h (LOADER_ADDRESS, LOADER_ARCH_CFLAGS...).
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(LOADER_ADDRESS 0x2000000000)
    set(HAS_POKEDATA_WORKAROUND TRUE)
elseif(ANDROID_ABI STREQUAL "x86_64")
    set(LOADER_ADDRESS 0x600000000000)
elseif(ANDROID_ABI STREQUAL "armeabi-v7a")
    set(LOADER_ADDRESS 0x20000000)
else()
    message(FATAL_ERROR "Unsupported ABI ${ANDROID_ABI}")
endif()

# ---- talloc ------------------------------------------------------------------------------------
add_library(talloc STATIC "${TALLOC_SRC}/talloc.c")
# replace.h is this project's minimal replacement of Samba's generated one.
target_include_directories(talloc PUBLIC "${TALLOC_SRC}")
target_compile_definitions(talloc PRIVATE
    TALLOC_BUILD_VERSION_MAJOR=2 TALLOC_BUILD_VERSION_MINOR=5 TALLOC_BUILD_VERSION_RELEASE=0)
set_target_properties(talloc PROPERTIES POSITION_INDEPENDENT_CODE ON)

# ---- loader ------------------------------------------------------------------------------------
add_executable(proot_loader "${PROOT_SRC}/loader/loader.c" "${PROOT_SRC}/loader/assembly.S")
target_include_directories(proot_loader PRIVATE "${PROOT_SRC}")
target_compile_definitions(proot_loader PRIVATE _FILE_OFFSET_BITS=64 _GNU_SOURCE)
target_compile_options(proot_loader PRIVATE -fPIC -ffreestanding -O2)
# --no-gc-sections: release builds add --gc-sections, which would drop pokedata_workaround (arm64),
# a symbol nothing references but that proot locates by offset at runtime.
target_link_options(proot_loader PRIVATE
    -static -nostdlib -fuse-ld=lld
    "LINKER:--build-id=none,--rosegment,--no-gc-sections,-Ttext=${LOADER_ADDRESS},-z,noexecstack")
set_target_properties(proot_loader PROPERTIES
    OUTPUT_NAME "proot-loader" PREFIX "lib" SUFFIX ".so" POSITION_INDEPENDENT_CODE OFF)

# ---- proot -------------------------------------------------------------------------------------
# build.h is generated by proot's makefile from compile checks; NDK targets support both features.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gen/build.h"
"#ifndef BUILD_H
#define BUILD_H
#undef VERSION
#define VERSION \"${PROOT_VERSION}\"
#define HAVE_PROCESS_VM
#define HAVE_SECCOMP_FILTER
#endif
")

set(PROOT_SOURCES

    execve/enter.c

    extension/ashmem_memfd/ashmem_memfd.c
    extension/extension.c
    extension/fake_id0/access.c
    extension/fake_id0/chmod.c
    extension/fake_id0/chown.c
    extension/fake_id0/chroot.c
    extension/fake_id0/exec.c
    extension/fake_id0/fake_id0.c
    extension/fake_id0/getsockopt.c
    extension/fake_id0/helper_functions.c
    extension/fake_id0/link.c
    extension/fake_id0/mk.c
    extension/fake_id0/open.c
    extension/fake_id0/rename.c
    extension/fake_id0/sendmsg.c
    extension/fake_id0/socket.c
    extension/fake_id0/stat.c
    extension/fake_id0/symlink.c
    extension/fake_id0/unlink.c
    extension/fake_id0/utimensat.c
    extension/fix_symlink_size/fix_symlink_size.c

list(TRANSFORM PROOT_SOURCES PREPEND "${PROOT_SRC}/")

if(HAS_POKEDATA_WORKAROUND)
    # Offset of the loader's pokedata_workaround symbol from its entry point (arm64 only).
    get_filename_component(TOOLCHAIN_BIN "${CMAKE_C_COMPILER}" DIRECTORY)
    set(LOADER_INFO_C "${CMAKE_CURRENT_BINARY_DIR}/gen/loader-info.c")
    add_custom_command(
        OUTPUT "${LOADER_INFO_C}"
        COMMAND "${CMAKE_COMMAND}"
            -DREADELF=${TOOLCHAIN_BIN}/llvm-readelf
            -DLOADER=$<TARGET_FILE:proot_loader>
            -DOUT=${LOADER_INFO_C}
            -P "${CMAKE_CURRENT_SOURCE_DIR}/loader-info.cmake"
        DEPENDS proot_loader "${CMAKE_CURRENT_SOURCE_DIR}/loader-info.cmake"
        VERBATIM)
    list(APPEND PROOT_SOURCES "${LOADER_INFO_C}")
endif()

add_executable(proot ${PROOT_SOURCES})
target_include_directories(proot PRIVATE "${PROOT_SRC}" "${CMAKE_CURRENT_BINARY_DIR}/gen")
target_compile_definitions(proot PRIVATE
    _FILE_OFFSET_BITS=64 _GNU_SOURCE ARG_MAX=131072
    "PROOT_UNBUNDLE_LOADER=\"/proot-loader-set-PROOT_LOADER\"")
# Some proot sources rely on <string.h> being pulled in transitively by glibc headers.
target_compile_options(proot PRIVATE -include string.h -Wall -Wextra -O2 -Wno-unused-parameter -Wno-sign-compare)
target_link_libraries(proot PRIVATE talloc)
target_link_options(proot PRIVATE "LINKER:-z,noexecstack")
set_target_properties(proot PROPERTIES OUTPUT_NAME "proot" PREFIX "lib" SUFFIX ".so")
